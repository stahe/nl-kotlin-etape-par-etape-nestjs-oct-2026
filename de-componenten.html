import androidx.compose.foundation.BorderStroke
import androidx.compose.foundation.background

import androidx.compose.foundation.layout.Column

import androidx.compose.material3.MaterialTheme

// de "stijl" van de kaart: constanten die PRIVÉ zijn voor dit bestand. Geen andere
// component kan ze zien, noch er last van hebben
private val initialsColor = Color(0xFF087EA4)

@Composable
fun DoctorCard(doctor: Doctor, rank: Int, modifier: Modifier = Modifier, compact: Boolean = false) {
    // een waarde berekend uit de parameters
    val initials = "${doctor.firstName.first()}${doctor.lastName.first()}"
    Card(
        modifier = modifier.padding(vertical = 4.dp),
        shape = RoundedCornerShape(12.dp),
        // een groene rand voor een beschikbare arts (null: geen rand)
        border = if (doctor.available) BorderStroke(2.dp, Color(0xFF4CAF50)) else null,
    ) {
        Row(Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
            // een gekleurde cirkel, met de initialen in het midden
            Box(
                contentAlignment = Alignment.Center,
                modifier = Modifier.size(40.dp).background(initialsColor, CircleShape),
            ) {
                Text(initials, color = Color.White, fontWeight = FontWeight.Bold)
            }
            Spacer(Modifier.width(12.dp))
            // weight(1f): de tekst neemt de volledige resterende breedte in
            Column(Modifier.weight(1f)) {
                Text("$rank. ${doctor.title} ${doctor.firstName} ${doctor.lastName}", style = MaterialTheme.typography.titleMedium)
                if (!compact) {
                    Text(doctor.specialty, color = Color.Gray)
                }
            }
        }
    }
}

    • regel 46: de parameters van de kaart.     • regel 48: een waarde die uit de parameters wordt berekend;

@Composable
fun DoctorsPage() {
    var doctors by remember {
        mutableStateOf(
            listOf(
                Doctor(1, "Mme", "Marie", "PELISSIER", "general practitioner", true),
                Doctor(2, "Mr", "Paul", "TATOU", "pediatrician", false),
                Doctor(3, "Melle", "Sophie", "DUPONT", "general practitioner", true),
            ),
        )
    }
    var compact by remember { mutableStateOf(false) }

    ExamplePage("Example 10 - components and their parameters") {
        ListItem(
            headlineContent = { Text("compact display") },
            trailingContent = { Switch(checked = compact, onCheckedChange = { compact = it }) },
        )
        // de ouder wijzigt ZIJN gegevens: Compose voert deze functie opnieuw uit, die
        // de kinderen opnieuw aanroept met hun nieuwe parameters
        OutlinedButton(onClick = { doctors = doctors.map { it.copy(available = !it.available) } }) {
            Text("Toggle availabilities")
        }
        Spacer(Modifier.height(8.dp))
        doctors.forEachIndexed { i, d ->
            // elke kaart ontvangt ZIJN parameters; key: zijn identiteit (zie voorbeeld 04)
            key(d.id) {
                DoctorCard(doctor = d, rank = i + 1, compact = compact)
            }
        }
        Spacer(Modifier.height(16.dp))
        Text("The same card, in compact version, by setting the parameter:", color = Color.Gray)
        // de ouder bepaalt ook de BREEDTE van het kind, via de parameter modifier
        DoctorCard(doctor = doctors[0], rank = 1, compact = true, modifier = Modifier.width(260.dp))
    }
}

import androidx.compose.runtime.setValue
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp

@Composable
fun Counter(label: String, max: Int, onIncrement: (Int) -> Unit, onLimit: () -> Unit) {
    // de toestand SPECIFIEK voor deze teller: elke Counter heeft zijn eigen
    var value by remember { mutableIntStateOf(0) }

    fun plus() {
        if (value >= max) {
            onLimit()
            return
        }
        value++
        // we INFORMEREN de ouder en geven hem de nieuwe waarde door
        onIncrement(value)
    }

    Row(verticalAlignment = Alignment.CenterVertically) {
        Text(label, Modifier.width(110.dp))
        OutlinedButton(onClick = { plus() }) { Text("+1") }
        Spacer(Modifier.width(12.dp))
        SuggestionChip(onClick = {}, label = { Text("$value / $max") })
    }
}
    • regel 30: het type van een functie wordt geschreven als [(parameters) -> result]: [(Int) -> Unit] ontvangt een [Int] en geeft niets terug; [() -> Unit] heeft geen parameters;
    • regel 32: de EIGEN toestand van de teller: elke [Counter] heeft er een eigen;
    • regels 34–41: wanneer de teller het maximum bereikt, informeert hij de ouder ([onLimit()]); anders verhoogt hij zichzelf en geeft hij zijn nieuwe waarde door ([onIncrement(value)]).

package fr.stahe.examples.ex11

import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.material3.OutlinedTextField
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier

@Composable
fun TextField(value: String, onChange: (String) -> Unit, label: String, max: Int, help: String = "") {
    val full = value.length == max
    OutlinedTextField(
        value = value,
        // het veld weigert de extra tekens: de toetsaanslag wordt alleen aan de
        // ouder doorgegeven als de tekst max tekens niet overschrijdt
        onValueChange = { text -> if (text.length <= max) onChange(text) },
        label = { Text(label) },
        singleLine = true,
        // de rand wordt rood wanneer het veld vol is
        isError = full,
        supportingText = { Text(if (full) "Maximum length reached - ${value.length}/$max" else "$help - ${value.length}/$max") },
        modifier = Modifier.fillMaxWidth(),
    )
}
    • regel 28: de toetsaanslag wordt alleen naar de ouder gestuurd als de tekst niet meer dan [max] tekens bevat: het veld „weigert” overtollige tekens;
    • regels 32–33: de rand wordt rood als het veld vol is, en de tooltip geeft dit aan.

@Composable
fun ParentPage() {
    var total by remember { mutableIntStateOf(0) }
    var message by remember { mutableStateOf("") }
    var firstName by remember { mutableStateOf("Marie") }
    var lastName by remember { mutableStateOf("PELISSIER") }

    // de functie die door een teller wordt aangeroepen: ze ontvangt de waarde die de teller doorgeeft
    fun onIncrement(label: String, value: Int) {
        total++
        message = "$label is now $value"
    }

    val titleStyle = MaterialTheme.typography.titleSmall
    ExamplePage("Example 11 - the child notifies its parent") {
        Card {
            Column(Modifier.padding(12.dp)) {
                Text("1. Functions called by the children", style = titleStyle)
                Counter(
                    label = "Coffees",
                    max = 3,
                    onIncrement = { v -> onIncrement("Coffees", v) },
                    onLimit = { message = "No more than 3 coffees!" },
                )
                Counter(
                    label = "Glasses of water",
                    max = 8,
                    onIncrement = { v -> onIncrement("Glasses of water", v) },
                    onLimit = { message = "Enough drinking." },
                )
                Spacer(Modifier.height(8.dp))
                Text("Total of clicks: $total - $message", fontStyle = FontStyle.Italic)
            }
        }
        Card {
            Column(Modifier.padding(12.dp)) {
                Text("2. Controlled components", style = titleStyle)
                TextField(value = firstName, onChange = { firstName = it }, label = "First name", max = 20, help = "Required")
                TextField(value = lastName, onChange = { lastName = it }, label = "Last name", max = 20, help = "Displayed in upper case")
                Spacer(Modifier.height(8.dp))
                Text(
                    buildAnnotatedString {
                        append("Data of the parent: ")
                        withStyle(SpanStyle(fontWeight = FontWeight.Bold)) { append("$firstName ${lastName.uppercase()}") }
                    },
                )
                Spacer(Modifier.height(8.dp))
                OutlinedButton(onClick = {
                    firstName = ""
                    lastName = ""
                }) { Text("Clear (from the parent)") }
            }
        }
    }
}

// een kolom van een tabel van T
class TableColumn<T>(val title: String, val weight: Float = 1f, val cell: @Composable (row: T, index: Int) -> Unit)

@Composable
fun <T> DataTable(
    rows: List<T>,
    columns: List<TableColumn<T>>,
    // getoond wanneer er geen rij is
    empty: @Composable () -> Unit = { Text("No data.") },
) {
    Column {
        // de rij met de titels
        Row {
            for (c in columns) {
                Text(c.title, Modifier.weight(c.weight).padding(6.dp), fontWeight = FontWeight.Bold)
            }
        }
        HorizontalDivider()
        if (rows.isEmpty()) {
            Box(Modifier.padding(6.dp)) { ProvideTextStyle(TextStyle(color = Color.Gray)) { empty() } }
        }
        rows.forEachIndexed { i, row ->
            if (i > 0) HorizontalDivider(color = Color(0xFFE0E0E0))
            Row(verticalAlignment = Alignment.CenterVertically) {
                // de cellen van DEZE rij, getoond door de functies van de ouder
                for (c in columns) {
                    Box(Modifier.weight(c.weight).padding(6.dp)) { c.cell(row, i) }
                }
            }
        }
    }
}

// ex13/Clock.kt - een component die start en stopt
// -------------------------------------------------------------------------
// Het leven van een component op het scherm ("de compositie"):
//   1. ze BETREEDT de compositie: haar functie wordt voor het
//      eerst uitgevoerd. Wat moet starten, start daar;
//   2. ze wordt zo vaak als nodig HERCOMPOSEERD (haar toestand of haar parameters
//      zijn veranderd);
//   3. ze VERLAAT de compositie: ze verdwijnt voorgoed van het scherm.
//      Wat was gestart, wordt daar GESTOPT.
// Compose levert "effecten" om op die momenten te handelen, buiten
// de weergave zelf:
//   - LaunchedEffect(key) { ... }: start een COROUTINE (code die kan
//     wachten, zie voorbeeld 19) bij het betreden van de compositie, start haar opnieuw
//     als key verandert en ANNULEERT haar bij het verlaten;
//   - DisposableEffect(key) { ...; onDispose { ... } }: code bij het
//     betreden, en de opruiming ervan (onDispose) bij het verlaten.
// Een klok start een timer wanneer ze verschijnt... en moet die
// absoluut stoppen wanneer ze verdwijnt: met LaunchedEffect gebeurt
// dat automatisch.
// =========================================================================

import androidx.compose.runtime.setValue

import kotlinx.coroutines.delay
import java.time.LocalTime
import java.time.format.DateTimeFormatter

fun currentTime(): String = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm:ss"))

@Composable
fun Clock(onLog: (String) -> Unit, color: Color = Color(0xFF3F51B5)) {
    var time by remember { mutableStateOf(currentTime()) }
    // de LAATST ontvangen functie onLog (de onderstaande effecten leven
    // langer dan één uitvoering van de functie)
    val log by rememberUpdatedState(onLog)

    // de compositie betreden en verlaten
    DisposableEffect(Unit) {
        log("Clock: entering the composition - timer started")
        onDispose { log("Clock: leaving the composition - timer stopped") }
    }

    // de timer: een coroutine, automatisch GEANNULEERD bij het verlaten
    LaunchedEffect(Unit) {
        while (true) {
            delay(1000) // een seconde wachten, zonder de telefoon te blokkeren
            time = currentTime()
        }
    }

    // een parameter die verandert: het effect wordt opnieuw gestart (behalve de eerste keer)
    var firstColor by remember { mutableStateOf(true) }
    LaunchedEffect(color) {
        if (firstColor) firstColor = false else log("Clock: recomposed with a new color")
    }

    Text(time, fontSize = 40.sp, fontFamily = FontFamily.Monospace, color = color)
}
    • regel 47: [rememberUpdatedState]: effecten duren langer dan één functieaanroep; ze moeten de LAATSTE ontvangen functie aanroepen ([onLog]), niet de eerste;
    • regels 50–52: betreden van de compositie en verlaten ([onDispose]);
    • regels 57–59: de timer, een oneindige lus... in een coroutine: [delay(1000)] wacht één seconde ZONDER de telefoon te blokkeren; de coroutine wordt geannuleerd bij het verlaten van de compositie, en de lus daarmee;
    • regel 65: een effect waarvan de sleutel een parameter is: het wordt opnieuw geactiveerd wanneer de kleur verandert.

@Composable
fun LifecyclePage() {
    var withClock by remember { mutableStateOf(true) }
    var red by remember { mutableStateOf(false) }
    val log = remember { mutableStateListOf<String>() }
    var search by remember { mutableStateOf("") }
    var showSearch by remember { mutableStateOf(false) }

    fun trace(message: String) {
        log.add("${currentTime()} $message")
    }

    // het leven van de activity: we ABONNEREN ons op haar toestandswijzigingen bij het betreden
    // van de compositie, we ZEGGEN HET ABONNEMENT OP bij het verlaten (onDispose)
    val lifecycle = LocalLifecycleOwner.current.lifecycle
    DisposableEffect(lifecycle) {
        val observer = LifecycleEventObserver { _, event -> trace("Activity: ${event.name}") }
        lifecycle.addObserver(observer)
        onDispose { lifecycle.removeObserver(observer) }
    }

    TaskTitle(if (search.isEmpty()) "Example 13" else "Search: $search")

    ExamplePage("Example 13 - the lifecycle") {
        Card {
            Column(Modifier.padding(12.dp)) {
                CheckboxRow("show the clock", withClock) { withClock = it }
                CheckboxRow("in red", red) { red = it }
                if (withClock) {
                    Clock(onLog = { trace(it) }, color = if (red) Color.Red else Color(0xFF3F51B5))
                }
            }
        }
        Card {
            Column(Modifier.padding(12.dp)) {
                if (!showSearch) {
                    OutlinedButton(onClick = { showSearch = true }) { Text("Search...") }
                } else {
                    // een veld dat de focus krijgt zodra het verschijnt
                    val focus = remember { FocusRequester() }
                    OutlinedTextField(
                        value = search,
                        onValueChange = { search = it },
                        placeholder = { Text("the title follows the input") },
                        modifier = Modifier.fillMaxWidth().focusRequester(focus),
                    )
                    LaunchedEffect(Unit) { focus.requestFocus() }
                }
            }
        }
        Text("Log", style = MaterialTheme.typography.titleSmall)
        log.forEachIndexed { i, line -> Text("${i + 1}. $line", fontSize = 12.sp) }
    }
}

val LocalClock = compositionLocalOf { LocalDateTime.of(2026, 1, 1, 0, 0) }

@Composable
fun ClockProvider(periodMs: Long = 1000, content: @Composable () -> Unit) {
    var now by remember { mutableStateOf(LocalDateTime.now()) }
    // als de periode verandert, wordt de coroutine geannuleerd en opnieuw gestart
    LaunchedEffect(periodMs) {
        while (true) {
            delay(periodMs)
            now = LocalDateTime.now()
        }
    }
    CompositionLocalProvider(LocalClock provides now, content = content)
}

@Composable
fun Highlighter(color: Color = Color(0xFFFFF3CD), content: @Composable () -> Unit) {
    var touched by remember { mutableStateOf(false) }
    // zweven van een muis: een "interaction source", waarvan de
    // toestand "hovered" wordt geobserveerd
    val interactions = remember { MutableInteractionSource() }
    val hovered by interactions.collectIsHoveredAsState()
    // een GEANIMEERDE kleurwijziging (200 ms)
    val fill by animateColorAsState(if (touched || hovered) color else Color.Transparent, tween(200), label = "fill")
    Box(
        Modifier
            .background(fill)
            .hoverable(interactions)
            // de aanrakingen van de vinger, op het laagste niveau: de vinger raakt aan,
            // daarna komt hij omhoog (of het gebaar wordt geannuleerd). requireUnconsumed = false:
            // ook als de inhoud (een knop) het indrukken al heeft afgehandeld
            .pointerInput(Unit) {
                awaitEachGesture {
                    awaitFirstDown(requireUnconsumed = false)
                    touched = true
                    waitForUpOrCancellation()
                    touched = false
                }
            },
    ) {
        content()
    }
}

@Composable
fun LongPressButton(durationMs: Int = 1000, onAction: () -> Unit, content: @Composable () -> Unit) {
    val progress = remember { Animatable(0f) }
    // de scope van de coroutines van de component: ze worden geannuleerd als ze verdwijnt
    val scope = rememberCoroutineScope()
    val action by rememberUpdatedState(onAction)

    Row(
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier.pointerInput(Unit) {
            detectTapGestures(
                // onPress: de vinger raakt aan; tryAwaitRelease WACHT tot hij omhoog komt
                onPress = {
                    val filling = scope.launch {
                        progress.animateTo(1f, tween(durationMs, easing = LinearEasing))
                        // de animatie is tot het einde doorgelopen: het indrukken duurde lang genoeg
                        progress.snapTo(0f)
                        action()
                    }
                    tryAwaitRelease()
                    // de vinger komt te vroeg omhoog: het vullen wordt gestopt, de ring loopt leeg
                    if (filling.isActive) {
                        filling.cancel()
                        scope.launch { progress.animateTo(0f, tween(200)) }
                    }
                },
            )
        },
    ) {
        CircularProgressIndicator(
            progress = { progress.value },
            modifier = Modifier.size(22.dp),
            color = Color.Red,
            strokeWidth = 3.dp,
            trackColor = Color(0xFFFFEBEE),
        )
        Spacer(Modifier.width(8.dp))
        content()
    }
}

// -------------------------------------------------------------------------
// Met Compose is een dialoogvenster (AlertDialog) een component zoals elke andere:
// het wordt getoond zolang een toestand erom vraagt. Om een vraag te stellen
// en op het antwoord te WACHTEN, schrijven we een kleine klasse, Confirmation, waarvan de
// functie confirm een SUSPEND-functie is (suspend, zie voorbeeld
// 19): ze opent het dialoogvenster en wacht dan, zonder de telefoon te blokkeren,
// tot de gebruiker antwoordt. De aanroepende code wordt dan heel eenvoudig,
// in een coroutine:
//
//     if (confirmation.confirm("Delete this doctor?")) { ... }
//
// Het sluiten van het dialoogvenster ZONDER te antwoorden (buiten tikken, de Terugknop van de telefoon)
// geldt als "nee".
// Het dialoogvenster zelf wordt getoond door ConfirmDialog(confirmation),
// dat één keer op het scherm wordt geplaatst.
// =========================================================================
package fr.stahe.examples.ex16

import androidx.compose.material3.AlertDialog
import androidx.compose.material3.Button
import androidx.compose.material3.ButtonDefaults
import androidx.compose.material3.Text
import androidx.compose.material3.TextButton
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.graphics.Color
import kotlinx.coroutines.CompletableDeferred

class Confirmation {
    // de gestelde vraag (null: geen dialoogvenster)
    var question by mutableStateOf<String?>(null)
        private set

    // het verwachte antwoord: een waarde die LATER zal aankomen
    private var answer: CompletableDeferred<Boolean>? = null

    suspend fun confirm(message: String): Boolean {
        val waiting = CompletableDeferred<Boolean>()
        answer = waiting
        question = message // het dialoogvenster wordt getoond
        try {
            return waiting.await() // de coroutine wordt opgeschort tot het antwoord er is
        } finally {
            question = null // het dialoogvenster verdwijnt
        }
    }

    fun respond(yes: Boolean) {
        answer?.complete(yes)
    }
}

// remember: de Confirmation overleeft de hercomposities van het scherm
@Composable
fun rememberConfirmation(): Confirmation = remember { Confirmation() }

@Composable
fun ConfirmDialog(confirmation: Confirmation) {
    // ?: return: geen vraag, niets te tonen
    val question = confirmation.question ?: return
    // AlertDialog: een Material-dialoogvenster, met titel, inhoud en knoppen
    AlertDialog(
        onDismissRequest = { confirmation.respond(false) }, // gesloten zonder antwoord: "nee"
        title = { Text("Confirmation") },
        text = { Text(question) },
        dismissButton = { TextButton(onClick = { confirmation.respond(false) }) { Text("No, go back") } },
        confirmButton = {
            Button(
                onClick = { confirmation.respond(true) },
                colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFD32F2F)),
            ) { Text("Yes, confirm") }
        },
    )
}
    • regels 36–37: de gestelde vraag ([null]: geen venster), een geobserveerde toestand;
    • regels 42–49: [CompletableDeferred]: een waarde die LATER zal aankomen. [confirm] toont het venster en WACHT dan op het antwoord ([await()]), zonder de telefoon te laten vastlopen; [finally]: het venster verdwijnt in elk geval;
    • regels 53–54: een knop van het venster LEVERT het antwoord: de wachtende coroutine gaat verder;

@Composable
fun ContactsPage() {
    val contacts = remember { mutableStateListOf(Contact(1, "Jean MARTIN"), Contact(2, "Alice DURAND"), Contact(3, "Luc GERMAN")) }
    var message by remember { mutableStateOf("") }
    val confirmation = rememberConfirmation()
    // het gebied voor de kortstondige berichten en een scope om coroutines te starten
    val messages = remember { SnackbarHostState() }
    val scope = rememberCoroutineScope()

    // een suspend-functie: ze "wacht" op het antwoord van de gebruiker
    suspend fun remove(c: Contact) {
        if (!confirmation.confirm("Delete the contact ${c.name}?")) {
            message = "Deletion abandoned."
            return
        }
        val position = contacts.indexOf(c)
        contacts.remove(c)
        message = "${c.name} has been deleted."
        // een kortstondig bericht onderaan het scherm; showSnackbar wacht tot het
        // verdwijnt en geeft aan of zijn knop is aangeraakt
        val result = messages.showSnackbar("${c.name} deleted", actionLabel = "Undo", duration = SnackbarDuration.Short)
        if (result == SnackbarResult.ActionPerformed) {
            contacts.add(position, c)
            message = "${c.name} is back."
        }
    }

    suspend fun removeAll() {
        if (confirmation.confirm("Delete the ${contacts.size} contacts?")) {
            contacts.clear()
            message = "All the contacts have been deleted."
        }
    }

    ExamplePage("Example 16 - a confirmation dialog", messages = messages) {
        for (c in contacts) {
            ListItem(
                leadingContent = { Icon(Icons.Default.Person, contentDescription = null) },
                headlineContent = { Text(c.name) },
                trailingContent = {
                    // launch: een coroutine, om de suspend-functie te kunnen aanroepen
                    IconButton(onClick = { scope.launch { remove(c) } }) {
                        Icon(Icons.Outlined.Delete, contentDescription = "delete ${c.name}", tint = Color.Red)
                    }
                },
            )
        }
        if (contacts.isEmpty()) {
            ListItem(headlineContent = { Text("No contact.", color = Color.Gray) })
        }
        Spacer(Modifier.height(12.dp))
        Button(
            onClick = { scope.launch { removeAll() } },
            // de knop is uitgeschakeld wanneer de lijst leeg is
            enabled = contacts.isNotEmpty(),
            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFD32F2F)),
        ) { Text("Delete all") }
        Spacer(Modifier.height(12.dp))
        if (message.isNotEmpty()) {
            Text(message, color = Color(0xFF3F51B5))
        }
    }
    // het dialoogvenster, wanneer er een vraag wordt gesteld
    ConfirmDialog(confirmation)
}
