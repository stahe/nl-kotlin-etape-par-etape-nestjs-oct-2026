    6. Asynchrone verwerking en gedeelde toestand

    6.1. Voorbeeld 19: coroutines, Flow en timers

    Veel dingen gebeuren „later”: tikken van een timer, toetsaanslagen, het antwoord van een server. Waar [Dart] zijn [Future] en [Stream] heeft, heeft [Kotlin] de COROUTINES:

    • Een functie [suspend] kan WACHTEN ([delay], een antwoord van een server...) zonder de telefoon te laten vastlopen: tijdens het wachten blijft het scherm actief. Ze geeft LATER ÉÉN waarde terug: dit is het equivalent van een functie [async] die een [Future] teruggeeft;
    • een [Flow<T>] is een REEKS waarden in de tijd, getransformeerd met operatoren ([map], [debounce]...): het equivalent van een [Stream];
    • een coroutine wordt gestart in een SCOPE ([LaunchedEffect], [rememberCoroutineScope], [viewModelScope]...); ze wordt GEANNULEERD wanneer haar scope eindigt.

// ex19/Ex19Activity.kt - coroutines, Flow, timers
// -------------------------------------------------------------------------
// Veel dingen gebeuren "later": de tikken van een timer, de
// toetsaanslagen, het antwoord van een server. Kotlin heeft hiervoor
// COROUTINES:
//   - een "suspend"-functie kan WACHTEN (delay, het antwoord van een
//     server...) zonder de telefoon te blokkeren: tijdens het wachten leeft het scherm
//     gewoon door. Ze geeft ÉÉN waarde terug, later;
//   - een Flow<T> is een REEKS waarden in de tijd (tikken,
//     invoer...), getransformeerd met operatoren (map, debounce...);
//   - een coroutine wordt gestart in een SCOPE: LaunchedEffect,
//     rememberCoroutineScope...; ze wordt geannuleerd wanneer haar scope verdwijnt.
// Compose kan een Flow tonen en hercomponeert zichzelf bij elke waarde:
// collectAsStateWithLifecycle.
// =========================================================================
@file:OptIn(FlowPreview::class, ExperimentalCoroutinesApi::class)

val doctors = listOf("Marie PELISSIER", "Paul TATOU", "Sophie DUPONT", "Jacques DURAND", "Martine PELLETIER")

// een gesimuleerde "server": een SUSPEND-functie, waarvan het antwoord na een willekeurige vertraging
// aankomt
suspend fun search(text: String): List<String> {
    delay(200L + Random.nextLong(600))
    return doctors.filter { it.lowercase().contains(text) }
}

// 1. een Flow die met de builder flow { } is gebouwd: elke emit STUURT een
// waarde naar wie haar verzamelt
fun countdown(start: Int): Flow<Int> = flow {
    for (remaining in start downTo 0) {
        emit(remaining)
        if (remaining > 0) delay(1000)
    }
}

fun currentTime(): String = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm:ss"))

class Ex19Activity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { ExamplesTheme { AsynchronousPage() } }
    }
}

@Composable
fun AsynchronousPage() {
    val log = remember { mutableStateListOf<String>() }
    fun trace(message: String) = log.add("${currentTime()} $message")

    // de Flow wordt ÉÉN keer aangemaakt (remember): zonder dit zou elke hercompositie
    // het aftellen opnieuw starten!
    val countdownFlow = remember { countdown(20) }
    // collectAsStateWithLifecycle: een toestand die de waarden van de Flow volgt
    // (null vóór de eerste); het verzamelen stopt als de applicatie
    // naar de achtergrond gaat
    val remaining by countdownFlow.collectAsStateWithLifecycle(initialValue = null)

    // 2. zoeken terwijl u typt
    var input by remember { mutableStateOf("") }
    var results by remember { mutableStateOf(listOf<String>()) }
    LaunchedEffect(Unit) {
        // snapshotFlow: een Flow van de opeenvolgende waarden van een Compose-toestand
        snapshotFlow { input }
            // de DEBOUNCE: een invoer wordt pas doorgegeven na 400 ms zonder toetsaanslag
            .debounce(400)
            .map { it.trim().lowercase() }
            // mapLatest: een nieuwe invoer ANNULEERT de vorige zoekopdracht,
            // als die niet klaar is: geen verouderd antwoord
            .mapLatest { text ->
                // minder dan twee letters: geen zoekopdracht, geen resultaat
                if (text.length < 2) return@mapLatest emptyList()
                trace("search sent: \"$text\"")
                try {
                    search(text)
                } catch (e: CancellationException) {
                    trace("search cancelled (stale): \"$text\"")
                    throw e // een geannuleerde coroutine moet geannuleerd blijven
                }
            }
            // collect: verwerkt de waarden van de Flow, een voor een
            .collect { results = it }
    }

    // 3. een waarde die na één seconde aankomt: produceState start een
    // coroutine die een toestand PRODUCEERT
    val message by produceState<String?>(initialValue = null) {
        delay(1000)
        value = "three (one second later)"
    }

    val titleStyle = MaterialTheme.typography.titleSmall
    ExamplePage("Example 19 - asynchronous code") {
        Card(Modifier.fillMaxWidth()) {
            Column(Modifier.padding(12.dp)) {
                Text("A countdown (a Flow, collected)", style = titleStyle)
                Text(remaining?.toString() ?: "...", style = MaterialTheme.typography.displaySmall)
            }
        }
        Card(Modifier.fillMaxWidth()) {
            Column(Modifier.padding(12.dp)) {
                Text("A search as you type (debounce, stale responses)", style = titleStyle)
                OutlinedTextField(
                    value = input,
                    onValueChange = { input = it },
                    placeholder = { Text("at least two letters: pel, dur...") },
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth(),
                )
                for (r in results) Text("• $r")
                if (results.isEmpty()) Text("no result", color = Color.Gray)
            }
        }
        Card(Modifier.fillMaxWidth()) {
            Column(Modifier.padding(12.dp)) {
                Text("Waiting for a value (produceState)", style = titleStyle)
                // zolang de waarde niet is aangekomen: de wachttekst
                if (message == null) Text("waiting...", color = Color.Gray) else Text("received: $message")
            }
        }
        Text("Log", style = titleStyle)
        log.forEachIndexed { i, line -> Text("${i + 1}. $line", fontSize = 12.sp) }
    }
}
    • regels 65–67: een gesimuleerde „server”: een suspend-functie waarvan het antwoord na een willekeurige vertraging binnenkomt;
    • regels 72–75: een [Flow] aangemaakt met [flow [ ... ]]: elke [emit] stuurt een waarde naar degene die verzamelt; [downTo]: een aflopende lus;
    • regels 96–100: de [Flow] wordt EENMALIG aangemaakt ([remember]): anders zou elke hercompositie het aftellen opnieuw starten! [collectAsStateWithLifecycle] maakt in feite een toestand aan die de waarden ervan volgt; het verzamelen stopt wanneer de app naar de achtergrond gaat;
    • regels 107–125: zoeken tijdens het typen. [snapshotFlow [ input ]]: de [Flow] van opeenvolgende waarden van een toestand van [Compose]; [debounce(400)], DEBOUNCE: een invoer wordt pas doorgestuurd na 400 ms zonder toetsaanslagen; [mapLatest]: een nieuwe invoer ANNULEERT de vorige zoekopdracht als die nog niet is voltooid: geen verouderd antwoord;
    • regels 119–121: een geannuleerde coroutine ontvangt een [CancellationException]: die kan worden vastgelegd, maar moet opnieuw worden gegooid ([throw e]) om de annulering te voltooien;
    • regels 130–132: [produceState]: een coroutine die een toestand PRODUCEERT (hier een waarde die na een seconde binnenkomt).

    Links bij het opstarten; rechts nadat snel „pel”, „du” en daarna „dup” is getypt: alleen de toetsaanslagen die door een pauze werden gevolgd, leidden tot een zoekopdracht, en het scherm toont het resultaat van de laatste.

    6.2. Voorbeeld 20: een gedeelde toestand

    Een „store” is een toestand die BUITEN de schermen leeft, als ÉÉN enkele instantie, gedeeld door iedereen die hem leest. Voorbeeld 14 deed dit handmatig; [Android] biedt hier het volgende voor: een object dat op verzoek wordt aangemaakt ([viewModel()]), vervolgens identiek wordt teruggegeven aan iedereen die erom vraagt, en dat opnieuw wordt aangemaakt bij een wijziging van de activity (schermrotatie, taalwijziging). Dit is het equivalent van de pakketten [provider] of [riverpod] van [Flutter].

// ex20/stores/Cart.kt - een GEDEELDE TOESTAND, in een ViewModel
// -------------------------------------------------------------------------
// Een "store" is een toestand die BUITEN de schermen leeft, in ÉÉN enkel
// exemplaar, gedeeld door allen die hem lezen. Voorbeeld 14 deed dit
// met de hand (CompositionLocal); Android biedt hiervoor het ViewModel:
//   - een object dat bij het EERSTE verzoek wordt aangemaakt en daarna identiek wordt teruggegeven aan
//     alle functies van de activity die erom vragen: viewModel();
//   - dat het opnieuw aanmaken van de activity OVERLEEFT (schermrotatie,
//     taalwijziging): het wordt pas vernietigd wanneer de activity definitief
//     wordt gesloten.
// Gebruikelijke organisatie:
//   - de TOESTAND: een PRIVÉ MutableStateFlow, een observeerbare waarde; daarbuiten wordt alleen
//     een alleen-leesversie ervan gezien (StateFlow);
//   - de ACTIES: functies, de enige die de toestand wijzigen;
//   - de berekende waarden: andere StateFlows, afgeleid van de eerste (map),
//     die ALLEEN veranderen als hun waarde verandert.
// =========================================================================
package fr.stahe.examples.ex20.stores

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.flow.update

data class Item(val label: String, val price: Double)

class Cart : ViewModel() {
    private val _items = MutableStateFlow(listOf<Item>())
    val items: StateFlow<List<Item>> = _items.asStateFlow()

    // AFGELEIDE waarden: stateIn maakt er StateFlows van, die leven zolang
    // het ViewModel (viewModelScope)
    val count: StateFlow<Int> = items.map { it.size }.stateIn(viewModelScope, SharingStarted.Eagerly, 0)
    val total: StateFlow<Double> = items.map { list -> list.sumOf { it.price } }.stateIn(viewModelScope, SharingStarted.Eagerly, 0.0)

    // update: vervangt de waarde door een nieuwe, berekend uit de oude
    fun add(item: Item) = _items.update { it + item }

    fun clear() {
        _items.value = emptyList()
    }
}
    • regels 34–35: de TOESTAND: een PRIVÉ [MutableStateFlow], een observeerbare waarde; naar buiten is alleen een alleen-lezenversie ([StateFlow]) zichtbaar;
    • regels 39–40: AFGELEIDE waarden: [map], daarna [stateIn], dat op zijn beurt objecten [StateFlow] aanmaakt die bestaan zolang [ViewModel] bestaat ([viewModelScope]). Een [StateFlow] zendt alleen uit als zijn waarde VERANDERT;
    • regels 43–46: de ACTIES zijn de enige die de toestand wijzigen; [update] vervangt de waarde door een nieuwe, berekend op basis van de oude.

// ex20/stores/Preferences.kt - een toestand die het stoppen overleeft
// -------------------------------------------------------------------------
// Om een gegeven het stoppen van de applicatie te laten overleven, moet het
// OP HET APPARAAT worden bewaard. De bibliotheek Jetpack DataStore biedt een klein
// sleutel -> waarde-woordenboek (Boolean, Int, Float, String...), specifiek voor
// de applicatie en bewaard in een bestand van de telefoon:
//   - lezen is een Flow: hij geeft de bewaarde waarden af (asynchroon
//     gelezen, zonder het scherm te blokkeren) en daarna elke nieuwe waarde;
//   - schrijven is een suspend-functie (edit).
// Bewaar er NOOIT gevoelige gegevens in (wachtwoord, token): ze worden
// in leesbare tekst weggeschreven.
// =========================================================================
package fr.stahe.examples.ex20.stores

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.booleanPreferencesKey
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.floatPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

// HET voorkeurenbestand van het voorbeeld: een extensie-eigenschap van
// Context, gedelegeerd aan preferencesDataStore (één enkele DataStore per bestand)
val Context.storage: DataStore<Preferences> by preferencesDataStore(name = "example20")

// wat is bewaard
data class Settings(val dark: Boolean, val size: Float)

class UserPreferences(private val storage: DataStore<Preferences>) {
    private val darkKey = booleanPreferencesKey("dark")
    private val sizeKey = floatPreferencesKey("size")

    // wat het apparaat heeft bewaard, anders de standaardwaarden
    val settings: Flow<Settings> = storage.data.map { p -> Settings(p[darkKey] ?: false, p[sizeKey] ?: 16f) }

    // één optionele benoemde parameter per voorkeur: alleen wat wordt opgegeven, wordt gewijzigd
    suspend fun update(dark: Boolean? = null, size: Float? = null) {
        storage.edit { p ->
            if (dark != null) p[darkKey] = dark
            if (size != null) p[sizeKey] = size
        }
    }
}
    • regel 28: HET voorkeurenbestand van het voorbeeld: een extensie-eigenschap van [Context], gedelegeerd aan [preferencesDataStore] (slechts één [DataStore] per bestand);
    • regel 38: het lezen is een [Flow]: die geeft de opgeslagen waarden (gelezen zonder het scherm te blokkeren) en daarna elke nieuwe waarde;
    • regels 41–44: het schrijven is een suspend-functie ([edit]). Gevoelige gegevens (wachtwoorden, tokens) worden daar NOOIT opgeslagen: ze zouden in platte tekst worden weggeschreven.

// VOEGT toe aan het gedeelde winkelmandje: de winkel LEEST het winkelmandje niet, hij
// wordt niet hercomponeerd wanneer het winkelmandje verandert
@Composable
fun Shop(cart: Cart = viewModel()) {
    Card {
        Column(Modifier.padding(12.dp)) {
            Text("Shop", style = MaterialTheme.typography.titleSmall)
            for (p in products) {
                TextButton(onClick = { cart.add(p) }) {
                    Icon(Icons.Default.Add, contentDescription = null)
                    Text("${p.label} (€${p.price})")
                }
            }
        }
    }
}

// LEEST hetzelfde winkelmandje: alleen het aantal en het totaal (alleen hercomponeerd
// als een van beide verandert)
@Composable
fun CartSummary(cart: Cart = viewModel()) {
    val count by cart.count.collectAsStateWithLifecycle()
    val total by cart.total.collectAsStateWithLifecycle()
    Card {
        Row(Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
            Icon(Icons.Default.ShoppingCart, contentDescription = null)
            Spacer(Modifier.width(8.dp))
            Text("$count item(s), €$total", Modifier.weight(1f))
            TextButton(onClick = { cart.clear() }, enabled = count > 0) { Text("Clear") }
        }
    }
}

// een LOKALE toestand (rememberSaveable), die met de component verdwijnt...
// tenzij ze in een SaveableStateProvider is geplaatst (zie Ex20Activity.kt)
@Composable
fun LocalCounter(title: String) {
    var clicks by rememberSaveable { mutableIntStateOf(0) }
    Card {
        ListItem(
            headlineContent = { Text(title) },
            trailingContent = { OutlinedButton(onClick = { clicks++ }) { Text("$clicks click(s)") } },
        )
    }
}

class Ex20Activity : ComponentActivity() {
    // lazy: het object wordt bij het eerste gebruik aangemaakt (de applicatiecontext
    // bestaat nog niet wanneer de activity wordt geconstrueerd)
    private val preferences by lazy { UserPreferences(applicationContext.storage) }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { App(preferences) }
    }
}

@Composable
fun App(preferences: UserPreferences) {
    // de bewaarde voorkeuren: null zolang ze niet zijn gelezen
    val settings by preferences.settings.collectAsStateWithLifecycle(initialValue = null)
    val s = settings ?: return // niets te tonen vóór het lezen (enkele milliseconden)
    val scope = rememberCoroutineScope()

    // het donkere thema, of het lichte
    ExamplesTheme(dark = s.dark) {
        // de tekstgrootte: fontScale vermenigvuldigt de grootte van ALLE teksten
        val density = LocalDensity.current
        CompositionLocalProvider(LocalDensity provides Density(density.density, fontScale = s.size / 16f)) {
            SharedStatePage(
                dark = s.dark,
                size = s.size,
                onDark = { v -> scope.launch { preferences.update(dark = v) } },
                onSize = { v -> scope.launch { preferences.update(size = v) } },
            )
        }
    }
}

@Composable
fun SharedStatePage(dark: Boolean, size: Float, onDark: (Boolean) -> Unit, onSize: (Float) -> Unit) {
    var areaVisible by rememberSaveable { mutableStateOf(true) }
    // de bewaarder van de toestanden van de verwijderde inhouden
    val keptStates = rememberSaveableStateHolder()

    ExamplePage("Example 20 - a shared state") {
        ListItem(
            headlineContent = { Text("dark theme") },
            trailingContent = { Switch(checked = dark, onCheckedChange = onDark) },
        )
        ListItem(
            headlineContent = { Text("text size: ${size.roundToInt()}") },
            supportingContent = { Slider(value = size, onValueChange = onSize, valueRange = 12f..22f, steps = 9) },
        )
        ListItem(
            modifier = Modifier.toggleable(value = areaVisible, onValueChange = { areaVisible = it }),
            leadingContent = { Checkbox(checked = areaVisible, onCheckedChange = null) },
            headlineContent = { Text("show the area") },
        )
        if (areaVisible) {
            // verwijderd: deze componenten verdwijnen, met hun lokale toestand
            Shop()
            CartSummary()
            LocalCounter("Local state")
            // ook verwijderd, maar zijn toestand wordt BEWAARD door keptStates, onder de sleutel "counter"
            keptStates.SaveableStateProvider("counter") {
                LocalCounter("Local state, kept by a SaveableStateHolder")
            }
        }
        Text(
            "Outside the area, a second summary of the SAME cart (it stays there when the area disappears):",
            Modifier.padding(vertical = 8.dp),
        )
        CartSummary()
    }
}
