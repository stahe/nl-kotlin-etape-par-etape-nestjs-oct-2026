    3.1.1. Het project

    Het voorbeeldproject is aangemaakt met [Android Studio] (sjabloon [Empty Activity], zie het vorige hoofdstuk) en daarna aangevuld. Een [Android]-project is een [Gradle]-project: het hulpmiddel dat de bibliotheken downloadt, de [Kotlin]-code compileert en het bestand [.apk] genereert dat op de telefoon wordt geïnstalleerd. Het wordt beschreven door enkele bestanden die in [Kotlin] zijn geschreven (de bestanden [.kts], „Kotlin script”).

    // de Compose-compiler: transformeert de @Composable-functies

    namespace = "fr.stahe.examples"
    // de versie van de Android-API die gebruikt wordt om te COMPILEREN
    compileSdk {
        version = release(37)
    }

    defaultConfig {
        // de UNIEKE identificatie van de applicatie op de telefoon
        applicationId = "fr.stahe.examples"
        // de oudste geaccepteerde Android-versie: Android 8.0 (API 26)
        minSdk = 26
        // de Android-versie waarvoor de applicatie getest is
        targetSdk = 37
        versionCode = 1
        versionName = "1.0"
        // added: het adres van de JSON-server (voorbeelden 22 tot en met 25), te wijzigen bij
        // het compileren: gradlew installDebug -Pserver=http://192.168.1.20:8080
        val server = providers.gradleProperty("server").getOrElse("http://localhost:8080")
        buildConfigField("String", "SERVER_URL", "\"$server\"")

        testInstrumentationRunner = "androidx.test.runner.AndroidJUnitRunner"
    }

        release {
            optimization {
                enable = false
            }
        }

        targetCompatibility = JavaVersion.VERSION_11

        buildConfig = true

@Composable
fun CalculatorPage() {
    // de toestand: wat de gebruiker heeft getypt of gekozen
    var label by rememberSaveable { mutableStateOf("Pen") }
    var priceText by rememberSaveable { mutableStateOf("2.5") } // de tekst van het veld, zoals getypt
    var quantity by rememberSaveable { mutableIntStateOf(1) }
    var vatRate by rememberSaveable { mutableDoubleStateOf(20.0) }
    // de historie van de laatste 5 wijzigingen van het totaal incl. btw: een ONWIJZIGBARE
    // lijst, bij elke wijziging vervangen door een nieuwe lijst
    var history by rememberSaveable { mutableStateOf(listOf<String>()) }

    // BEREKENDE waarden: gewone lokale variabelen, bij
    // elke uitvoering van de functie opnieuw berekend
    val priceExclVat = price(priceText)
    val totalExclVat = quantity * priceExclVat
    val vatAmount = totalExclVat * vatRate / 100
    val inclVat = totalExclVat + vatAmount

    // ALLE wijzigingen van de toestand lopen via deze lokale functie: ze
    // noteert het totaal incl. btw vóór de wijziging, past de wijziging toe en
    // vult de historie aan als het totaal is veranderd
    fun change(update: () -> Unit) {
        val old = inclVat
        update()
        // de toestandsvariabelen hebben hun nieuwe waarde AL
        val newTotal = totalInclVat(price(priceText), quantity, vatRate)
        if (newTotal != old) {
            history = (listOf("${euros.format(old)} -> ${euros.format(newTotal)}") + history).take(5)
        }
    }

    // de titel van de applicatie VOLGT de toestand, net als al het andere
    TaskTitle("Incl. VAT: ${euros.format(inclVat)}")

        Card {

                // een GECONTROLEERD veld: het toont value en roept onValueChange aan
                // bij elke toetsaanslag, met de nieuwe tekst

                    value = label,
                    // het wijzigen van het artikel zet de hoeveelheid terug op 1: het gebeurt op de plek waar
                    // we weten WAAROM de toestand verandert
                    onValueChange = { newValue ->

@Composable
fun TasksPage() {
    // door Compose GEOBSERVEERDE lijsten: het toevoegen, verwijderen of vervangen van een
    // element veroorzaakt de hercompositie. (remember, en niet rememberSaveable:
    // ze gaan verloren als de activity opnieuw wordt aangemaakt, zie voorbeeld 20)
    val tasks = remember {
        mutableStateListOf(Task(1, "Install Android Studio", done = true), Task(2, "Read the chapter on events"))
    }
    val log = remember { mutableStateListOf<String>() }
    var input by remember { mutableStateOf("") }
    var nextId by remember { mutableIntStateOf(3) }
    // de FocusRequester: een manier om vanuit de code de focus aan het veld te geven (de cursor en het
    // toetsenbord van de telefoon)
    val focus = remember { FocusRequester() }

    fun trace(message: String) {
        val time = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        log.add(0, "$time $message")
    }

    // aangeroepen door de knop EN door de "OK"-toets van het toetsenbord (keyboardActions)
    fun add() {
        val label = input.trim()
        if (label.isEmpty()) {
            return
        }
        tasks.add(Task(nextId++, label))
        trace("added \"$label\"")
        input = ""
        // de focus keert terug naar het veld: de volgende taak kan worden getypt
        focus.requestFocus()
    }

    fun toggle(task: Task, done: Boolean) {
        // we VERVANGEN de taak door een gewijzigde kopie
        tasks[tasks.indexOf(task)] = task.copy(done = done)
        trace("\"${task.label}\": ${if (done) "done" else "to do"}")
    }

    fun remove(task: Task, how: String) {
        tasks.remove(task)
        trace("removed \"${task.label}\" ($how)")
    }

    // de schermdichtheid: om pixels om te zetten in dp
    val density = LocalDensity.current.density

    ExamplePage("Example 04 - events") {
        Row(verticalAlignment = Alignment.CenterVertically) {
            OutlinedTextField(
                value = input,
                onValueChange = { input = it },
                placeholder = { Text("new task (Escape to clear)") },
                singleLine = true,
                // de "OK"-toets van het toetsenbord van de telefoon voegt de taak toe
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(onDone = { add() }),
                modifier = Modifier
                    .weight(1f)
                    .focusRequester(focus)
                    // een fysiek toetsenbord (emulator, tablet): Escape wist het veld
                    .onPreviewKeyEvent { event ->
                        if (event.key == Key.Escape) {
                            input = ""
                            true // de gebeurtenis is afgehandeld
                        } else {
                            false // de gebeurtenis gaat verder op zijn weg
                        }
                    },
            )
            Spacer(Modifier.width(8.dp))
            Button(onClick = { add() }) { Text("Add") }
        }
        Spacer(Modifier.height(12.dp))
        for (t in tasks) {
            // key(t.id): de IDENTITEIT van de regel. De toestand van elke regel (zijn
            // veeg) volgt ZIJN taak, ook als de lijst verandert (zie de cursus)
            key(t.id) {
                TaskRow(
                    task = t,
                    onToggle = { done -> toggle(t, done) },
                    onRemove = { how -> remove(t, how) },
                )
            }
        }
        Spacer(Modifier.height(12.dp))
        // pointerInput: elke component wordt gevoelig voor gebaren
        Box(
            contentAlignment = Alignment.Center,
            modifier = Modifier
                .fillMaxWidth()
                .height(80.dp)
                .background(Color(0xFFE7F5FF), RoundedCornerShape(8.dp))
                .pointerInput(Unit) {
                    detectTapGestures(
                        // de positie van de aanraking, relatief ten opzichte van de component (in pixels)
                        onPress = { p -> trace("press at (${(p.x / density).roundToInt()}, ${(p.y / density).roundToInt()})") },
                        onDoubleTap = { trace("double tap") },
                        onLongPress = { trace("long press") },
                    )
                },
        ) {
            Text("Touch this area (tap, double tap, long press)")
        }
        Spacer(Modifier.height(16.dp))
        Text("Log", style = MaterialTheme.typography.titleSmall)
        for (line in log) {
            Text(line, fontSize = 12.sp)
        }
    }
}

// de naam van elk veld van het formulier
enum class Field { TITLE, FIRSTNAME, LASTNAME, AGE, NEWSLETTER, EMAIL, COUNTRY, PRESENTATION }

// de gegevens van het formulier: een ONWIJZIGBAAR object (alles is val). Om het
// te "wijzigen" maken we er een kopie van: profile.copy(lastName = "DUPONT"). De
// functie copy wordt door Kotlin voor elke "data"-klasse geschreven
data class Profile(
    val title: String,
    val firstName: String,
    val lastName: String,
    val age: Int,
    val newsletter: Boolean,
    val email: String,
    val country: String,
    val presentation: String,
) {
    // de waarde van een veld, aangeduid met zijn naam
    fun value(f: Field): Any = when (f) {
        Field.TITLE -> title
        Field.FIRSTNAME -> firstName
        Field.LASTNAME -> lastName
        Field.AGE -> age
        Field.NEWSLETTER -> newsletter
        Field.EMAIL -> email
        Field.COUNTRY -> country
        Field.PRESENTATION -> presentation
    }

    // een kopie waarin ÉÉN veld, aangeduid met zijn naam, een nieuwe waarde heeft
    fun withValue(f: Field, v: Any): Profile = when (f) {
        Field.TITLE -> copy(title = v as String)
        Field.FIRSTNAME -> copy(firstName = v as String)
        Field.LASTNAME -> copy(lastName = v as String)
        Field.AGE -> copy(age = v as Int)
        Field.NEWSLETTER -> copy(newsletter = v as Boolean)
        Field.EMAIL -> copy(email = v as String)
        Field.COUNTRY -> copy(country = v as String)
        Field.PRESENTATION -> copy(presentation = v as String)
    }

    // het JSON-object van het profiel: één JSON-veld per veld van het formulier
    fun toJson(): JsonObject = JsonObject(
        Field.entries.associate { f ->
            val v = value(f)
            f.name.lowercase() to when (v) {
                is Int -> JsonPrimitive(v) // "is": een typetest; v is daarna een Int
                is Boolean -> JsonPrimitive(v)
                else -> JsonPrimitive(v.toString())
            }
        },
    )
}

// de volledige toestand: de waarden en de reeds verlaten velden
data class State(val values: Profile, val touched: Set<Field>)

// de mogelijke acties. "sealed": er zijn geen andere dan de
// hier gedeclareerde, de compiler weet dat
sealed interface ProfileAction {
    data class Update(val field: Field, val value: Any) : ProfileAction
    data class Leave(val field: Field) : ProfileAction
    data object Uppercase : ProfileAction // een uniek object: de actie heeft geen gegevens
    data object Reset : ProfileAction
}

// de reducer: hij wijzigt de ontvangen toestand NOOIT, hij geeft een nieuwe terug.
// In elke tak van de when heeft action het type van de tak ("smart cast"):
// action.field is toegankelijk voor een Update
fun reducer(state: State, action: ProfileAction): State = when (action) {
    is ProfileAction.Update -> state.copy(values = state.values.withValue(action.field, action.value))
    is ProfileAction.Leave -> state.copy(touched = state.touched + action.field)
    ProfileAction.Uppercase -> state.copy(
        values = state.values.copy(firstName = state.values.firstName.uppercase(), lastName = state.values.lastName.uppercase()),
    )
    ProfileAction.Reset -> State(initial, emptySet())
}

// geeft voor elk veld met een fout zijn bericht terug. buildMap: we vullen
// een wijzigbare Map, de functie geeft een ONWIJZIGBARE Map terug
fun validate(a: Account): Map<Field, String> = buildMap {
    if (a.title !in titles) put(Field.TITLE, "Choose a title from the list.")
    when {
        a.firstName.isBlank() -> put(Field.FIRSTNAME, "The first name is required.")
        a.firstName.length > 20 -> put(Field.FIRSTNAME, "The first name must not exceed 20 characters.")
    }
    when {
        a.lastName.isBlank() -> put(Field.LASTNAME, "The last name is required.")
        a.lastName.length > 20 -> put(Field.LASTNAME, "The last name must not exceed 20 characters.")
    }
    when {
        a.login.isEmpty() -> put(Field.LOGIN, "The login is required.")
        !loginFormat.matches(a.login) -> put(Field.LOGIN, "Unaccented lower-case letters, digits, dot, hyphen, underscore.")
    }
    when {
        a.password.isEmpty() -> put(Field.PASSWORD, "The password is required.")
        a.password.length < PASSWORD_MIN -> put(Field.PASSWORD, "At least $PASSWORD_MIN characters.")
    }
    // een regel die TWEE velden vergelijkt
    if (a.confirmation != a.password) put(Field.CONFIRMATION, "The confirmation does not match the password.")
}

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf

// typealias: een betekenisvollere naam voor een functietype
typealias Rule = (String) -> String?

// FABRIEKEN van regels: elk geeft een validatiefunctie terug
fun required(message: String): Rule = { v -> if (v.isBlank()) message else null }

// een regel die een andere AANVULT: "?:" (de Elvis-operator) geeft het
// resultaat van other, als dat niet null is, anders dat van de rest
fun maximum(max: Int, other: Rule): Rule = { v -> other(v) ?: if (v.length > max) "$max characters at most." else null }

class FormField(initialValue: String = "", private val rule: Rule = { null }) {
    // de toestand van het veld: geobserveerd door Compose, alleen HIER wijzigbaar (private set)
    var value by mutableStateOf(initialValue)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // heeft de gebruiker het veld al gewijzigd?
    private var modified = false

    // een toetsaanslag: het veld valideert zichzelf zodra de gebruiker het heeft gewijzigd
    fun change(newValue: String) {
        value = newValue
        modified = true
        error = rule(value)
    }

    fun validate(): Boolean {
        error = rule(value)
        return error == null
    }

    // opnieuw valideren, alleen als de gebruiker het veld al heeft gewijzigd
    fun revalidate() {
        if (modified) validate()
    }
}

// vararg: een willekeurig aantal velden, Form(firstName, lastName, login...)
class Form(private vararg val fields: FormField) {
    // eerst map (ALLE velden worden gevalideerd en tonen hun fout),
    // daarna all (zijn ze allemaal in orde?)
    fun validate(): Boolean = fields.map { it.validate() }.all { it }
}

fun timeRange(s: Slot, part: Part = Part.ALL): String = when (part) {
    Part.START -> time(s.startHour, s.startMinute)
    Part.END -> time(s.endHour, s.endMinute)
    Part.ALL -> "${time(s.startHour, s.startMinute)} - ${time(s.endHour, s.endMinute)}"
}

// EXTENSIONS van het type String: nieuwe eigenschappen voor ALLE
// strings. In de body verwijst this naar de string
// "marie pelissier" -> "Marie Pelissier"; \p{L}: een letter, met of zonder accent
val String.capitalized: String
    get() = Regex("\\p{L}+").replace(this) { m ->
        m.value.lowercase().replaceFirstChar { it.uppercase() }
    }

// "Marie PELISSIER" -> "MP"
val String.initials: String
    get() = split(Regex("\\s+")).filter { it.isNotEmpty() }.joinToString("") { it.first().uppercase() }
    • regels 24–30: een OPTIONELE parameter, met zijn standaardwaarde ([part = Part.ALL]); [padStart(2, '0')]: „8:05” in plaats van „8:5”;
