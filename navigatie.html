    5. Navigatie

    Een applicatie heeft meerdere schermen. Bij [Android] wordt met [Compose] niet per scherm een activity geopend: ÉÉN activity geeft de schermen van de applicatie om beurten weer, aangestuurd door een ROUTER. Dit is de bibliotheek [navigation-compose] van [Jetpack], het equivalent van het pakket [go_router] voor [Flutter].

    // =========================================================================
// ex17/Routes.kt - de ROUTES van de applicatie
// -------------------------------------------------------------------------
// Een applicatie heeft meerdere schermen ("pagina's"). De ROUTER koppelt een
// ROUTE aan elk scherm. De bibliotheek Jetpack Navigation (navigation-
// compose) beheert een STAPEL van schermen: een scherm wordt erop gelegd (navigate),
// de Terug-knop van de telefoon haalt het eraf (popBackStack).
// Een route is een Kotlin-object, @Serializable:
//   - een "object" voor een scherm zonder parameters (Home);
//   - een "data class" voor een scherm met parameters: DoctorDetail(id = 2).
// Het type van de parameters wordt bij het compileren gecontroleerd: DoctorDetail("2")
// zou worden geweigerd. (Navigation kan deze objecten in de stapel bewaren, vandaar
// @Serializable.)
// Elk scherm kan ook een ADRES hebben, een "deep link"
// zoals app://examples/doctors/2: dat is het dichtst bij webadressen
// dat Android komt (zie Ex17Activity.kt).
// =========================================================================
package fr.stahe.examples.ex17

import kotlinx.serialization.Serializable

@Serializable
object Home

@Serializable
object DoctorList

// een routeparameter
@Serializable
data class DoctorDetail(val id: Int)

// een OPTIONELE parameter (met een standaardwaarde): ?q=... in het adres
@Serializable
data class Search(val q: String = "")

@Serializable
object About

// een oud adres, dat DOORVERWIJST naar de lijst met artsen
@Serializable
object Physicians

// het scherm voor onbekende adressen
@Serializable
data class NotFoundPage(val address: String)

// het begin van alle adressen van dit voorbeeld
const val BASE = "app://examples"

Het type van de parameters wordt bij het compileren gecontroleerd ([DoctorDetail("2")] zou worden geweigerd); [@Serializable]: [Navigation] weet hoe deze objecten op de stapel worden opgeslagen en hersteld als de activity opnieuw wordt aangemaakt;
    • regel 34: een optionele parameter met een standaardwaarde;
    • regel 48: elk scherm kan een ADRES hebben, een „deep link” van de vorm [app://examples/doctors/2]: dit komt bij [Android] het dichtst in de buurt van webadressen.

    // =========================================================================
// ex17/Ex17Activity.kt - de NavHost: de routes en hun schermen
// -------------------------------------------------------------------------
//   - rememberNavController(): de navigatiecontroller (de stapel);
//   - NavHost(nav, startDestination = Home) { ... }: het scherm van de
//     huidige route. Het blok beschrijft de navigatie-GRAAF: één
//     vermelding composable<Route> { ... } per route;
//   - entry.toRoute<DoctorDetail>(): het routeobject van het scherm, met
//     zijn parameters;
//   - deepLinks: het adres van het scherm. Dankzij het intent filter van
//     de activity (AndroidManifest.xml) kan een adres app://examples/...
//     zelfs van buiten de applicatie worden geopend:
//         adb shell am start -a android.intent.action.VIEW -d "app://examples/doctors/2"
// =========================================================================
package fr.stahe.examples.ex17

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController
import androidx.navigation.navDeepLink
import androidx.navigation.toRoute
import fr.stahe.examples.common.ExamplesTheme
import fr.stahe.examples.ex17.pages.AboutPage
import fr.stahe.examples.ex17.pages.HomePage
import fr.stahe.examples.ex17.pages.DoctorDetailPage
import fr.stahe.examples.ex17.pages.DoctorListPage
import fr.stahe.examples.ex17.pages.SearchPage
import fr.stahe.examples.ex17.pages.NotFoundPage as NotFoundScreen

class Ex17Activity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { ExamplesTheme { App() } }
    }
}

@Composable
fun App() {
    val nav = rememberNavController()
    Shell(nav) {
        NavHost(navController = nav, startDestination = Home) {
            composable<Home>(deepLinks = listOf(navDeepLink<Home>(basePath = "$BASE/home"))) {
                HomePage(onLink = { nav.openLink(it) }, onRedirect = { nav.navigate(Physicians) })
            }
            composable<DoctorList>(deepLinks = listOf(navDeepLink<DoctorList>(basePath = "$BASE/doctors"))) {
                // navigate: het scherm van de arts wordt BOVENOP de lijst gelegd
                DoctorListPage(onView = { id -> nav.navigate(DoctorDetail(id)) })
            }
            // app://examples/doctors/2: de parameter id wordt uit het adres gehaald
            composable<DoctorDetail>(deepLinks = listOf(navDeepLink<DoctorDetail>(basePath = "$BASE/doctors"))) { entry ->
                val route = entry.toRoute<DoctorDetail>()
                DoctorDetailPage(
                    id = route.id,
                    // popBackStack: het scherm eraf halen (zoals de Terug-knop); als
                    // er niets onder ligt (scherm geopend via een adres), naar de lijst gaan
                    onBack = { if (!nav.popBackStack()) nav.navigate(DoctorList) },
                    // het huidige scherm VERVANGEN (zonder push): de huidige arts
                    // wordt eraf gehaald (inclusive) voordat de volgende erop komt
                    onNext = { next ->
                        nav.navigate(DoctorDetail(next)) { popUpTo<DoctorDetail> { inclusive = true } }
                    },
                )
            }
            // app://examples/search?q=pediatrician
            composable<Search>(deepLinks = listOf(navDeepLink<Search>(basePath = "$BASE/search"))) { entry ->
                SearchPage(q = entry.toRoute<Search>().q, onSearch = { input -> nav.navigate(Search(input)) })
            }
            composable<About> { AboutPage() }
            // een DOORVERWIJZING: het scherm toont niets, het vervangt onmiddellijk
            // zijn eigen route door een andere
            composable<Physicians> {
                LaunchedEffect(Unit) {
                    nav.navigate(DoctorList) { popUpTo<Physicians> { inclusive = true } }
                }
            }
            composable<NotFoundPage> { entry ->
                NotFoundScreen(address = entry.toRoute<NotFoundPage>().address, onHome = { nav.goTo(Home) })
            }
        }
    }
}
    • regel 34: [import ... as NotFoundScreen]: een alias, omdat de schermfunctie [NotFoundPage] dezelfde naam heeft als de route [NotFoundPage];
    • regels 46–48: [rememberNavController()]: de navigatiecontroller (de stapel); [NavHost(nav, startDestination = Home)] toont het scherm van de huidige route;
    • regels 49–54: één item [composable<Route>] per route, met zijn adres ([deepLinks]). De schermen kennen de router niet: ze ontvangen FUNCTIES om te navigeren ([onLink], [onView]...);
    • regels 57–58: [entry.toRoute<DoctorDetail>()]: het routeobject, met zijn parameters. Met [navDeepLink<DoctorDetail>(basePath = ...)] wordt de parameter [id] uit het adres gehaald ([.../doctors/2]);
    • regel 63: [popBackStack()] haalt het scherm van de stapel, net als de knop Terug, en geeft [false] terug als er niets onder ligt (scherm geopend via een adres);
    • regels 66–67: het huidige scherm VERVANGEN: de huidige arts ([popUpTo<DoctorDetail> [ inclusive = true ]]) wordt van de stapel gehaald voordat de volgende erop wordt gezet;
    • regels 78–80: een DOORVERWIJZING: het scherm toont niets; het vervangt onmiddellijk zijn route door een andere. Navigatie is een „effect” ([LaunchedEffect]): u navigeert nooit TIJDENS het weergeven;

// een tabblad: de route waarnaar het leidt, zijn label, zijn pictogram en de test die
// zegt of het actief is voor een bepaald scherm
class Tab(val route: Any, val label: String, val icon: ImageVector, val active: (NavDestination) -> Boolean)

val tabs = listOf(
    Tab(Home, "Home", Icons.Default.Home) { it.hasRoute<Home>() },
    // het tabblad Doctors blijft actief op het scherm van een arts
    Tab(DoctorList, "Doctors", Icons.Default.Person) { it.hasRoute<DoctorList>() || it.hasRoute<DoctorDetail>() },
    Tab(Search("general practitioner"), "Search", Icons.Default.Search) { it.hasRoute<Search>() },
    Tab(About, "About", Icons.Default.Info) { it.hasRoute<About>() },
)

// het adres van het huidige scherm, om het te tonen
fun addressOf(entry: NavBackStackEntry?): String {
    val d = entry?.destination ?: return ""
    return when {
        d.hasRoute<Home>() -> "/"
        d.hasRoute<DoctorList>() -> "/doctors"
        d.hasRoute<DoctorDetail>() -> "/doctors/${entry.toRoute<DoctorDetail>().id}"
        d.hasRoute<Search>() -> "/search?q=${entry.toRoute<Search>().q}"
        d.hasRoute<About>() -> "/about"
        d.hasRoute<NotFoundPage>() -> entry.toRoute<NotFoundPage>().address
        else -> "?"
    }
}

// naar een tabblad gaan: de stapel wordt VERVANGEN door die van dit tabblad
fun NavHostController.goTo(route: Any) {
    navigate(route) {
        // alles wordt eraf gehaald, tot het eerste scherm (uitgezonderd)...
        popUpTo(graph.findStartDestination().id)
        // ... en hetzelfde scherm wordt niet twee keer erop gelegd
        launchSingleTop = true
    }
}

// een LINK (een adres) volgen; een onbekend adres leidt naar NotFoundPage
fun NavHostController.openLink(address: String) {
    try {
        navigate(Uri.parse("$BASE$address"))
    } catch (e: IllegalArgumentException) {
        navigate(NotFoundPage(address))
    }
}

@Composable
fun Shell(nav: NavHostController, content: @Composable () -> Unit) {
    val entry by nav.currentBackStackEntryAsState()
    val destination = entry?.destination
    Scaffold(
        topBar = { TopAppBar(title = { Text("Example 17 - the router") }) },
        // NavigationBar: de tabbalk onderaan, typisch voor een mobiel apparaat
        bottomBar = {
            NavigationBar {
                for (t in tabs) {
                    NavigationBarItem(
                        selected = destination != null && t.active(destination),
                        onClick = { nav.goTo(t.route) },
                        icon = { Icon(t.icon, contentDescription = null) },
                        label = { Text(t.label) },
                    )
                }
            }
        },
    ) { insets ->
        Column(Modifier.padding(insets).fillMaxSize()) {
            Text(
                "current address: ${addressOf(entry)}",
                fontFamily = FontFamily.Monospace,
                fontSize = 12.sp,
                modifier = Modifier.fillMaxWidth().background(Color(0xFFF5F5F5)).padding(8.dp),
            )
            Box(Modifier.padding(16.dp).weight(1f)) { content() }
        }
    }
}

@Composable
fun HomePage(onLink: (String) -> Unit, onRedirect: () -> Unit) {
    TaskTitle("Home")
    Column {
        Text(
            "Welcome. Touch the tabs: the screen changes, and so does its address. The phone's Back " +
                "button pops the pushed screens. An address can also be opened from " +
                "outside the application (cf. the course).",
        )
        Spacer(Modifier.height(16.dp))
        // links naar bepaalde adressen
        FlowRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            OutlinedButton(onClick = onRedirect) { Text("/physicians (redirection)") }
            OutlinedButton(onClick = { onLink("/unknown-page") }) { Text("Broken link") }
        }
    }
}

@Composable
fun DoctorListPage(onView: (Int) -> Unit) {
    TaskTitle("Doctors")
    Column {
        for (d in doctors) {
            ListItem(
                headlineContent = { Text("${d.title} ${d.firstName} ${d.lastName}") },
                trailingContent = { Icon(Icons.AutoMirrored.Filled.KeyboardArrowRight, contentDescription = null) },
                // de arts wordt erop GELEGD; de Terug-knop keert hier terug
                modifier = Modifier.clickable { onView(d.id) },
            )
        }
    }
}

// een pagina met een parameter: de router geeft haar id door, een Int
@Composable
fun DoctorDetailPage(id: Int, onBack: () -> Unit, onNext: (Int) -> Unit) {
    // find: de eerste overeenkomende arts, of null
    val doctor = doctors.find { it.id == id }
    if (doctor == null) {
        Text("Doctor no. $id does not exist.", color = Color.Red)
        return
    }
    val next = doctor.id % doctors.size + 1
    TaskTitle("${doctor.firstName} ${doctor.lastName}")
    Column {
        Text("${doctor.title} ${doctor.firstName} ${doctor.lastName}", style = MaterialTheme.typography.titleLarge)
        Text("Specialty: ${doctor.specialty}")
        Spacer(Modifier.height(16.dp))
        FlowRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            OutlinedButton(onClick = onBack) { Text("Back") }
            Button(onClick = { onNext(next) }) { Text("Next doctor") }
        }
    }
}

// =========================================================================
// een pagina die door haar route wordt aangestuurd: Search(q = "pediatrician"). Zoeken
// betekent NAAR de route Search(q = <input>) GAAN: de pagina toont wat de
// route zegt en niets anders (de Terug-knop keert terug naar de vorige
// zoekopdracht).
// =========================================================================
@Composable
fun SearchPage(q: String, onSearch: (String) -> Unit) {
    TaskTitle("Search")
    val results = doctors.filter { "${it.firstName} ${it.lastName} ${it.specialty}".lowercase().contains(q.lowercase()) }
    // de invoer begint met de waarde van de route; rememberSaveable(q):
    // als de route verandert (Terug-knop), begint de invoer opnieuw met de nieuwe waarde
    var input by rememberSaveable(q) { mutableStateOf(q) }
    Column {
        OutlinedTextField(
            value = input,
            onValueChange = { input = it },
            placeholder = { Text("name or specialty") },

package fr.stahe.examples.ex18

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue

enum class Role { ADMIN, DOCTOR, USER }

data class User(val login: String, val role: Role)

// drie "hard-coded" accounts (wachtwoord = login)
private val accounts = mapOf("admin" to Role.ADMIN, "pelissier" to Role.DOCTOR, "martin" to Role.USER)

object Session {
    // de toestand: null, niemand is aangemeld. private set: alleen hier wijzigbaar
    var user by mutableStateOf<User?>(null)
        private set

    fun logIn(login: String, password: String): Boolean {
        val role = accounts[login]
        if (role == null || password != login) {
            return false
        }
        user = User(login, role)
        return true
    }

    fun logOut() {
        user = null
    }
}

// ex18/Routes.kt - door guards BESCHERMDE routes
// -------------------------------------------------------------------------
// Een GUARD is een functie die de aangemelde gebruiker (of null) ontvangt en
// teruggeeft:
//   - null: het scherm mag worden getoond;
//   - een route: het scherm wordt door deze route VERVANGEN (een doorverwijzing).
// De component Protected(...) past de guards van een scherm toe. Omdat ze de
// gebruiker van de sessie LEEST (een geobserveerde toestand), wordt ze bij elke
// aanmelding of afmelding opnieuw geëvalueerd: een scherm dat niet meer is toegestaan, wordt verlaten.
//
// Het is het equivalent aan clientzijde van de NestJS-guards (JwtAuthGuard,
// RolesGuard) - met een FUNDAMENTEEL verschil: een client-guard is
// slechts een gemak. De applicatie is in handen van de gebruiker, die
// haar naar believen kan wijzigen. De echte bescherming van de gegevens blijft
// die van de SERVER (zie het hoofdstuk over de blackbox).
// =========================================================================
package fr.stahe.examples.ex18

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.navigation.NavHostController
import kotlinx.serialization.Serializable

// denied: door de guard requireRole achtergelaten wanneer de rol niet past
@Serializable
data class Home(val denied: String? = null)

// redirect: het adres van het scherm dat de gebruiker wilde openen
@Serializable
data class Login(val redirect: String? = null)

@Serializable
object Schedule

@Serializable
object Administration

const val BASE = "app://examples18"

typealias Guard = (User?) -> Any?

// de guard "aanmelding vereist": niet aangemeld -> naar de aanmeldpagina,
// met onthouden van het adres van het gevraagde scherm
fun requireLogin(address: String): Guard = { u -> if (u == null) Login(redirect = address) else null }

// de guard "rol vereist": aangemeld, maar niet met de juiste rol -> terug naar
// de startpagina, met een reden
fun requireRole(address: String, vararg roles: Role): Guard = { u ->
    if (u == null || u.role !in roles) Home(denied = address) else null
}

// de guards, in volgorde: de eerste die een route teruggeeft, wint
@Composable
fun Protected(nav: NavHostController, vararg guards: Guard, content: @Composable () -> Unit) {
    val redirection = guards.firstNotNullOfOrNull { guard -> guard(Session.user) }
    if (redirection == null) {
        content()
    } else {
        // navigeren is een "effect": we navigeren niet TIJDENS het tonen
        LaunchedEffect(redirection) {
            val here = nav.currentBackStackEntry?.destination?.id ?: return@LaunchedEffect
            // het geweigerde scherm wordt (inclusive) door de doorverwijzing vervangen
            nav.navigate(redirection) { popUpTo(here) { inclusive = true } }
        }
    }
}
    • regel 41: een GUARD is een functie die de gebruiker (of [null]) als argument neemt en [null] teruggeeft (het scherm mag worden weergegeven) of een ROUTE (het scherm wordt door deze route vervangen);
    • regels 45–50: twee guardfuncties: „aanmelden vereist” verwijst door naar het aanmeldscherm en onthoudt het gevraagde adres; „rol vereist” verwijst door naar het welkomstscherm met een reden;
    • regels 55–64: [Protected] past de guards toe voor een scherm: [firstNotNullOfOrNull], het eerste resultaat dat niet [null] is. Omdat het de sessie LEEST, wordt het bij elke aanmelding of afmelding opnieuw geëvalueerd: een scherm dat niet meer is toegestaan, wordt verlaten.

@Composable
fun App() {
    val nav = rememberNavController()
    Shell(nav) {
        NavHost(navController = nav, startDestination = Home()) {
            composable<Home> { entry -> HomePage(denied = entry.toRoute<Home>().denied) }
            composable<Login> { entry ->
                LoginPage(
                    redirection = entry.toRoute<Login>().redirect,
                    onLoggedIn = { r ->
                        // alleen een adres van DEZE applicatie wordt geaccepteerd
                        // de stapel wordt leeggemaakt (popUpTo de hele graaf): de aanmeldpagina
                        // blijft er niet in staan, Terug opent haar niet opnieuw
                        val clearStack = navOptions { popUpTo(nav.graph.id) { inclusive = true } }
                        if (r != null && r.startsWith("$BASE/")) {
                            nav.navigate(Uri.parse(r), clearStack)
                        } else {
                            nav.navigate(Home(), clearStack)
                        }
                    },
                )
            }
            composable<Schedule>(deepLinks = listOf(navDeepLink<Schedule>(basePath = "$BASE/schedule"))) {
                Protected(nav, requireLogin("$BASE/schedule")) { SchedulePage() }
            }
            composable<Administration>(deepLinks = listOf(navDeepLink<Administration>(basePath = "$BASE/administration"))) {
                Protected(nav, requireLogin("$BASE/administration"), requireRole("/administration", Role.ADMIN)) {
                    AdministrationPage()
                }
            }
        }
    }
}

// de aanmelding. Na een geslaagde aanmelding ontvangt onLoggedIn het adres van
// het scherm dat de gebruiker wilde openen (parameter redirect achtergelaten door de guard),

@Composable
fun LoginPage(redirection: String?, onLoggedIn: (String?) -> Unit) {
    var login by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var error by remember { mutableStateOf(false) }

    fun validate() {
        if (!Session.logIn(login.trim(), password)) {
            error = true
            return
        }
        onLoggedIn(redirection)
    }

    Column {
        if (redirection != null) {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFF8E1))) {
                Text("Log in to open ${redirection.removePrefix(BASE)}.", Modifier.padding(12.dp))
            }
        }
        OutlinedTextField(value = login, onValueChange = { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.fillMaxWidth())
        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = { Text("password") },
            singleLine = true,
            visualTransformation = PasswordVisualTransformation(),
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Go),
            keyboardActions = KeyboardActions(onGo = { validate() }),
            modifier = Modifier.fillMaxWidth(),
        )
        Spacer(Modifier.height(16.dp))
        Button(onClick = { validate() }) { Text("Log in") }
        if (error) {
            Text("Incorrect login or password.", color = Color.Red, modifier = Modifier.padding(top = 8.dp))
        }
    }
}
